160 return data[i];

162
164 void push_back(uint8_t pin, bool value) noexcept {
165 if (count < kMaxPins) {
166 data[count++] = {pin, value};

173 [[nodiscard]] bool contains(uint8_t pin) const noexcept {
174 for (uint8_t i = 0; i < count; ++i) {
175 if (data[i].first == pin) return true;
176 }
177 return false;
178 }
179
182 [[nodiscard]] const std::pair<uint8_t, bool>* find(uint8_t pin) const noexcept {
183 for (uint8_t i = 0; i < count; ++i) {
184 if (data[i].first == pin) return &data[i];
185 }
186 return nullptr;
187 }
188
189 // -- Iterator support for range-based for --------------------------------
190 using iterator = typename std::array<std::pair<uint8_t, bool>, kMaxPins>::const_iterator;
191 [[nodiscard]] iterator begin() const noexcept { return data.begin(); }
